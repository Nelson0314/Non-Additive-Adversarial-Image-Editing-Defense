```python
"""E31 判準補完：編輯輸出的感知劣化階梯。

為什麼需要這一支。E25 把防禦成功的判準由 `net_lpips` 改為語意軸是對的，
但 arXiv:2512.14320 定義的 ISR 是一個聯集——「語意不符 prompt」**或**
「明顯的感知劣化」。本專案只取了前半。在 strength=0.5 的全域 SDEdit 下輸出
主要由 prompt 重新生成，語意不符在原理上幾乎不可達成；文獻上真正被達成的
是後半（PhotoGuard 推向灰色目標，成功的樣子就是輸出糊掉）。

指標與門檻都不得憑聲譽選。作法沿用 E20 的四臂等 LPIPS 探針與 E28 的色度
階梯：造一道已知強度的劣化階梯，四個候選 NR 指標各評一次，再由人眼比對頁
判定哪一級開始「這張圖已經不能用」，門檻定在那一級。

執行：python scripts/p11_degrade_ladder.py
成本：本機 CPU 或 4 GB GPU，數分鐘。不需要 SD 權重。
"""

import argparse
import csv
import sys
from pathlib import Path

import torch
from PIL import Image
import torchvision.transforms as T

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))

from src.purify.ops import gaussian_blur, gaussian_noise, jpeg_real, quantize_real

# 候選指標。四個都是無參考的：判準要問的是「這張輸出本身還能不能用」，
# 不是「它離未防禦的輸出多遠」——後者正是 E25/E29 兩次判定失效的量。
CANDIDATES = ("niqe", "musiq", "clipiqa", "topiq_nr")
# 方向：越大越好者為 True。NIQE 是統計距離，越小越好。
HIGHER_IS_BETTER = {"niqe": False, "musiq": True, "clipiqa": True, "topiq_nr": True}


def ladder_arms():
    """四個算子各四級，由弱到強。強度值沿用 `src/purify/ops.py::eval_sweep`
    的量級，但四級之間的間距放大，因為此處要跨越「還能用」到「不能用」，
    而非量測淨化的殘存率。
    """
    return (
        [("blur", s) for s in (0.5, 1.0, 2.0, 4.0)]
        + [("jpeg", q) for q in (80, 50, 30, 10)]
        + [("noise", s) for s in (0.02, 0.05, 0.10, 0.20)]
        + [("quantize", n) for n in (64, 16, 8, 4)]
    )


def apply_arm(x, kind, strength):
    """階梯一律用真實實作，不用可微代理：這裡沒有梯度要走。"""
    if kind == "blur":
        return gaussian_blur(x, strength)
    if kind == "noise":
        return gaussian_noise(x, strength, seed=0)
    if kind == "jpeg":
        return jpeg_real(x, int(strength))
    if kind == "quantize":
        return quantize_real(x, int(strength))
    raise ValueError(f"未知的階梯算子 {kind!r}")


def load_sources(root: Path, size: int, device):
    """來源取既有 run 的**未防禦編輯輸出**，不是原圖。

    理由：判準作用的對象是編輯輸出。編輯輸出本身已經是擴散生成的，其 NR
    品質基線與自然照片不同，用原圖定錨會定在錯的基線上。
    """
    paths = sorted(root.glob("runs/e29*_tau0.10/*/edit_orig.png"))
    if not paths:
        raise FileNotFoundError(
            "找不到任何 edit_orig.png。本階梯必須以既有的未防禦編輯輸出為"
            "來源，請先確認 runs/e29c_*_tau0.10/ 已入版控"
        )
    out = []
    for p in paths:
        img = Image.open(p).convert("RGB").resize((size, size), Image.LANCZOS)
        out.append((f"{p.parent.parent.name}/{p.parent.name}",
                    T.ToTensor()(img).unsqueeze(0).to(device)))
    return out


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--out", default="runs/p11_degrade_ladder")
    ap.add_argument("--size", type=int, default=512)
    args = ap.parse_args()

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    out = ROOT / args.out
    out.mkdir(parents=True, exist_ok=True)

    import pyiqa
    metrics = {k: pyiqa.create_metric(k, device=device) for k in CANDIDATES}
    sources = load_sources(ROOT, args.size, device)
    print(f"[p11] device={device} 來源 {len(sources)} 張")

    rows = []
    for src_name, x in sources:
        base = {k: float(m(x.clamp(0, 1))) for k, m in metrics.items()}
        rows.append({"source": src_name, "kind": "none", "strength": 0.0,
                     "level": 0, **base})
        (out / "img").mkdir(exist_ok=True)
        _save(x, out / "img" / f"{_slug(src_name)}__none.png")
        for kind, strength in ladder_arms():
            level = [s for k, s in ladder_arms() if k == kind].index(strength) + 1
            y = apply_arm(x, kind, strength).clamp(0, 1)
            row = {"source": src_name, "kind": kind, "strength": strength,
                   "level": level,
                   **{k: float(m(y)) for k, m in metrics.items()}}
            rows.append(row)
            _save(y, out / "img" / f"{_slug(src_name)}__{kind}_{strength}.png")
            print(f"  {src_name} {kind}={strength}  "
                  + "  ".join(f"{k}={row[k]:.3f}" for k in CANDIDATES), flush=True)

    with open(out / "ladder.csv", "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=list(rows[0].keys()))
        w.writeheader()
        w.writerows(rows)
    _write_compare_page(out, rows)
    print(f"[p11] 寫出 {out/'ladder.csv'} 與 {out/'compare.html'}")
    print("[p11] 下一步由人眼判讀 compare.html，決定哪一級開始『不能用』，"
          "並選出在該級上可分的指標。門檻寫進 docs/RESULTS_E31.md")


def _slug(s):
    return s.replace("/", "__")


def _save(x, path):
    from torchvision.utils import save_image
    save_image(x.clamp(0, 1), path)


def _write_compare_page(out: Path, rows):
    """比對頁：每一列一個算子，欄為四級，格內附四個指標值。

    指標之間出現矛盾時以人眼為準，這是本專案第四次用同一個作法
    （E20 的 p1、E25 的 p5、E26 的 p7、E28 的 p10）。
    """
    srcs = sorted({r["source"] for r in rows})
    html = ["<!doctype html><meta charset='utf-8'>",
            "<title>E31 劣化階梯</title>",
            "<style>body{font-family:sans-serif;background:#111;color:#eee}"
            "table{border-collapse:collapse}td{padding:4px;text-align:center}"
            "img{width:180px;display:block}small{font-size:11px;color:#aaa}"
            "</style>",
            "<h1>E31 劣化階梯：哪一級開始『這張圖不能用』</h1>",
            "<p>來源是既有 run 的未防禦編輯輸出。請逐列由左至右看，"
            "指出第一個你認為已經不能用的欄位。</p>"]
    for s in srcs:
        html.append(f"<h2>{s}</h2><table>")
        base = [r for r in rows if r["source"] == s and r["kind"] == "none"][0]
        html.append("<tr><td>原始<img src='img/%s__none.png'><small>%s</small></td></tr>"
                    % (_slug(s), _fmt(base)))
        for kind in ("blur", "jpeg", "noise", "quantize"):
            html.append(f"<tr><td>{kind}</td>")
            for r in [r for r in rows if r["source"] == s and r["kind"] == kind]:
                html.append(
                    "<td><img src='img/%s__%s_%s.png'>"
                    "<small>%s<br>%s</small></td>"
                    % (_slug(s), kind, r["strength"], r["strength"], _fmt(r)))
            html.append("</tr>")
        html.append("</table>")
    (out / "compare.html").write_text("\n".join(html), encoding="utf-8")


def _fmt(r):
    return " ".join(f"{k}={r[k]:.2f}" for k in CANDIDATES)


if __name__ == "__main__":
    main()
```

```python
判準（規格 §4，源自 arXiv:2512.14320）：

    ISR = 語意失敗 OR 感知劣化

語意失敗沿用 E25 §1.2 的定義（配對差、n≥2、|mean| > sd，只用 SigLIP——
CLIP 的 edit_effect 對照未通過，見 E25 §1.1）。感知劣化是新增的那一半：
編輯輸出本身的無參考品質相對未防禦編輯下降超過門檻。

為什麼值得重判既有資料。`results.csv` 自 E2 起就有 `edit_niqe_a` 與
`edit_niqe_b`，全專案沒有任何腳本讀過它們——這與 E25 發現 `edit_siglip_*`
從未被讀過是同一型的疏漏。實測 runs/e29c_P_tau0.10 的 identity 那一格是
3.104 → 4.675（NIQE 越低越好），即防禦後的輸出確實較差。那個差值算不算
「不能用」，由 p11 的人眼階梯定門檻。

執行：python scripts/p12_isr_rejudge.py [--degrade_tau 1.0]
成本：只讀 CSV，秒級。
"""

import argparse
import csv
import json
import sys
from pathlib import Path
from statistics import mean, pstdev

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))


def judge_cell(deltas_sem, deltas_deg, degrade_tau):
    """一格（一個 run × 一個淨化臂）的 ISR 判定。

    `deltas_sem`：逐影像的 SigLIP(防禦後編輯) − SigLIP(未防禦編輯)。
    `deltas_deg`：逐影像的「劣化量」，已依指標方向轉為「越大越劣化」。
    `degrade_tau`：劣化門檻，由 p11 的人眼階梯定出。

    n≥2 是必要條件而非保險：n=1 時 pstdev 恆為 0，`|mean| > sd` 對任何負值
    自動成立，E25 曾因此在步數掃描的兩個 run 上產生 24 格假陽性。
    """
    if not deltas_sem or not deltas_deg:
        raise ValueError(
            "ISR 判定需要至少一筆配對差。空清單不得回傳 False——"
            "那會把「沒有資料」與「判定為未擋下」混為一談"
        )
    if len(deltas_sem) != len(deltas_deg):
        raise ValueError(
            f"兩軸的樣本數不符（{len(deltas_sem)} vs {len(deltas_deg)}），"
            "它們必須來自同一批影像"
        )
    n = len(deltas_sem)
    m_sem, sd_sem = mean(deltas_sem), pstdev(deltas_sem)
    m_deg, sd_deg = mean(deltas_deg), pstdev(deltas_deg)
    semantic_fail = bool(n >= 2 and m_sem < 0 and abs(m_sem) > sd_sem)
    degrade_fail = bool(n >= 2 and m_deg > degrade_tau)
    return {
        "n": n,
        "mean_dsiglip": m_sem, "sd_dsiglip": sd_sem,
        "mean_ddegrade": m_deg, "sd_ddegrade": sd_deg,
        "semantic_fail": semantic_fail,
        "degrade_fail": degrade_fail,
        "isr": bool(semantic_fail or degrade_fail),
    }


def collect(run_dir: Path, degrade_metric: str):
    """回傳 {(purify, strength, noise_split): (sem_deltas, deg_deltas)}。

    CSV 一律以 utf-8 讀：欄位 `stop_reason` 內含中文，Windows 預設 cp950
    會 UnicodeDecodeError。
    """
    path = run_dir / "results.csv"
    if not path.exists():
        return {}
    with open(path, encoding="utf-8") as f:
        rows = list(csv.DictReader(f))
    a_key, b_key = f"edit_{degrade_metric}_a", f"edit_{degrade_metric}_b"
    if rows and a_key not in rows[0]:
        raise KeyError(
            f"{path} 沒有 {a_key} 欄。該 run 早於該欄位加入，"
            f"不可靜默略過——請在報告中列為『無此軸的資料』"
        )
    out = {}
    for r in rows:
        if not r.get("edit_siglip_a") or not r.get("edit_siglip_b"):
            continue
        key = (r["purify"], r["strength"], r["noise_split"])
        sem = float(r["edit_siglip_b"]) - float(r["edit_siglip_a"])
        # NIQE 越低越好，故「劣化量」= b − a。若日後改用越高越好的指標
        # （musiq / clipiqa / topiq_nr），此處的符號必須跟著翻。
        deg = float(r[b_key]) - float(r[a_key])
        out.setdefault(key, ([], []))
        out[key][0].append(sem)
        out[key][1].append(deg)
    return out


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--runs", default="runs")
    ap.add_argument("--out", default="runs/p12_isr_rejudge")
    ap.add_argument("--degrade_metric", default="niqe",
                    help="results.csv 中已有的 NR 指標名。目前只有 niqe")
    ap.add_argument("--degrade_tau", type=float, default=1.0,
                    help="感知劣化門檻，由 p11 的人眼階梯定出。"
                         "預設 1.0 只是佔位，正式判定必須傳入定錨值")
    args = ap.parse_args()

    runs_root = ROOT / args.runs
    out = ROOT / args.out
    out.mkdir(parents=True, exist_ok=True)

    rows = []
    for run_dir in sorted(p for p in runs_root.iterdir() if p.is_dir()):
        cells = collect(run_dir, args.degrade_metric)
        for (purify, strength, split), (sem, deg) in sorted(cells.items()):
            r = judge_cell(sem, deg, args.degrade_tau)
            rows.append({"run": run_dir.name, "purify": purify,
                         "strength": strength, "noise_split": split, **r})

    with open(out / "isr.csv", "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=list(rows[0].keys()))
        w.writeheader()
        w.writerows(rows)

    n_isr = sum(r["isr"] for r in rows)
    n_sem = sum(r["semantic_fail"] for r in rows)
    n_deg = sum(r["degrade_fail"] for r in rows)
    lines = [
        "# E31 ISR 重判（既有 run）",
        "",
        f"- 判準：語意失敗 OR 感知劣化（{args.degrade_metric}，"
        f"門檻 {args.degrade_tau}）",
        f"- 格數：{len(rows)}",
        f"- 語意失敗：{n_sem}",
        f"- 感知劣化：{n_deg}",
        f"- **ISR 成立：{n_isr}**",
        "",
        "無淨化（purify=identity、strength=0.0）那一列是乾淨攻擊，",
        "即正式判準所在的位置。",
        "",
    ]
    clean = [r for r in rows if r["purify"] == "identity"
             and r["strength"] == "0.0"]
    lines.append("| run | split | n | ΔSigLIP | Δ劣化 | 語意 | 劣化 | ISR |")
    lines.append("|---|---|---|---|---|---|---|---|")
    for r in clean:
        lines.append(
            f"| {r['run']} | {r['noise_split']} | {r['n']} | "
            f"{r['mean_dsiglip']:+.4f} | {r['mean_ddegrade']:+.3f} | "
            f"{'是' if r['semantic_fail'] else '否'} | "
            f"{'是' if r['degrade_fail'] else '否'} | "
            f"{'**是**' if r['isr'] else '否'} |")
    (out / "summary.md").write_text("\n".join(lines) + "\n", encoding="utf-8")
    print(f"[p12] {len(rows)} 格，語意 {n_sem}、劣化 {n_deg}、ISR {n_isr}")
    print(f"[p12] 寫出 {out/'isr.csv'} 與 {out/'summary.md'}")


if __name__ == "__main__":
    main()
```

```python
**這是預估不是量測。** 最佳化在 τ=0.28 下會找到與「把 τ=0.10 的解放大」

from PIL import Image

ROOT = Path(__file__).resolve().parent.parent

from src.metrics.chroma import local_chroma_bias
from src.metrics.local_acutance import local_acutance_dev


def find_scale(measure, target, lo=1.0, hi=64.0, tol=1e-3, max_iter=40):
    """對單調遞增的 `measure` 二分搜尋，使其達到 `target`。

    上界不足時拋出而非回傳 hi：靜默回傳會讓呼叫端以為達到了 target，
    於是後續量到的 acut / chroma 對應的其實是別的 LPIPS，整份預估失效。
    """
    if measure(lo) >= target:
        return lo
    if measure(hi) < target:
        raise ValueError(
            f"上界 hi={hi} 只達到 {measure(hi):.4f}，未達 target={target}。"
            "請提高 hi，不可回傳上界充數"
        )
    for _ in range(max_iter):
        mid = 0.5 * (lo + hi)
        if measure(mid) < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return 0.5 * (lo + hi)

    ap.add_argument("--cell", default="runs/e29c_P_tau0.10/car_00__P__r16")
    ap.add_argument("--out", default="runs/p13_budget_probe")
    ap.add_argument("--targets", default="0.10,0.15,0.20,0.28")
    args = ap.parse_args()

    import piq

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    cell = ROOT / args.cell
    out = ROOT / args.out
    (out / "img").mkdir(parents=True, exist_ok=True)

    def load(name):
        img = Image.open(cell / name).convert("RGB")
        return T.ToTensor()(img).unsqueeze(0).to(device)

    x = load("orig.png")
    x_base = load("baseline_phi0.png")
    x_def = load("defended.png")
    delta = x_def - x_base
    lpips = piq.LPIPS().to(device)

    def build(k):
        return (x_base + k * delta).clamp(0, 1)

    rows = []
    for target in [float(t) for t in args.targets.split(",")]:
        k = find_scale(lambda k: float(lpips(build(k), x)), target)
        xd = build(k)
        d = (xd - x_base).abs()
        rows.append({
            "target_lpips": target,
            "scale": round(k, 4),
            "lpips": float(lpips(xd, x)),
            "acut": float(local_acutance_dev(x_base, xd)),
            "chroma": float(local_chroma_bias(x_base, xd)),
            "rms": float((d ** 2).mean().sqrt()),
            "linf": float(d.max()),
            "frac_gt_16_255": float((d > 16 / 255).float().mean()),
            "psnr": float(piq.psnr(xd, x, data_range=1.0)),
            # gate 的兩道門檻，直接把判定寫進 CSV，不留給讀者心算
            "acut_over_tau": float(local_acutance_dev(x_base, xd)) > 0.04,
            "chroma_over_tau": float(local_chroma_bias(x_base, xd)) > 0.8,
        })
        from torchvision.utils import save_image
        save_image(xd, out / "img" / f"scaled_{target}.png")
        print(f"  target={target}  k={k:.3f}  acut={rows[-1]['acut']:.4f}  "
              f"chroma={rows[-1]['chroma']:.4f}  rms={rows[-1]['rms']:.4f}  "
              f"psnr={rows[-1]['psnr']:.2f}", flush=True)

    with open(out / "probe.csv", "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=list(rows[0].keys()))
        w.writeheader()
        w.writerows(rows)

    html = ["<!doctype html><meta charset='utf-8'><title>E31 預算探針</title>",
            "<style>body{font-family:sans-serif;background:#111;color:#eee}"
            "img{width:280px}td{padding:6px;text-align:center}</style>",
            "<h1>把 τ=0.10 的解放大到各個 LPIPS 預算</h1>",
            "<p>這是沿既有解的射線放大，不是重新最佳化。用途是判斷 "
            "acut（門檻 0.04）與 chroma（門檻 0.8）在該量級是否翻上來。</p>",
            "<table><tr>"]
    for r in rows:
        html.append(
            "<td><img src='img/scaled_%s.png'><br>LPIPS=%.3f<br>"
            "acut=%.4f%s<br>chroma=%.4f%s<br>PSNR=%.2f dB</td>"
            % (r["target_lpips"], r["lpips"], r["acut"],
               " ⚠" if r["acut_over_tau"] else "", r["chroma"],
               " ⚠" if r["chroma_over_tau"] else "", r["psnr"]))
```

```bash
#
# 為什麼要重校。E29 定出的 site P lr=0.03 是在 τ_lpips=0.05 下校的。E31 把
```

```bash
#!/bin/bash
# E31 R2：正對照搜尋的主網格。設計見
# docs/specs/2026-08-02-e31-positive-control.md §5。
#
# 只跑加性臂 site P。目的不是比較加性與非加性——E29 已證實兩臂在現行運作點
# 都是零，兩個零之間的比較沒有內容。目的是找到**任何一個**擋得下編輯的
# 運作點，為量測裝置建立本專案從未有過的正對照。
#
# 三個軸各自對應一個文獻依據：
#   目標函數 —— untargeted 是 E2–E29 一路用的；targeted 是 PhotoGuard 的
#     diffusion attack 形式（arXiv:2302.06588），推向灰色目標，成功的樣子是
#     輸出劣化；suppress 是 cross-attention 免疫那條線（arXiv:2509.10359、
#     2512.14333）的著力點。後兩者從未在真實 SD 上產生過資料。
#   τ_lpips —— 0.10 保住與 E27–E29 的可比性；0.28 進文獻區間（DCT-Shield
#     自報 0.267、PhotoGuard/MIST/AdvDM/SDS/DiffusionGuard 0.284–0.362）。
#   strength —— 0.5 是本專案現行的威脅模型；0.3 對齊 PhotoGuard 的 SDEdit
#     評測。strength 同時決定防禦訓練與攻擊，白盒下兩者必須一致。
#
# 成本：12 格 × 2 圖。全鏈格約 100 步 × 2.36 s + 33.7 s 評測 ≈ 4.5 分／圖，
# crossattn 格較低。合計約 1.5–2 小時（H100、TF32 開）。
#
# 前置：scripts/drivers/e31_calibration.sh 的綁定者判定全部是 LPIPS hinge。
# LR_028 由該輪定出，執行前必須改成實際值。
#
# 用法：
#   bash scripts/drivers/e31_grid.sh
#   PY=/path/to/python LR_028=0.1 bash scripts/drivers/e31_grid.sh
set -euo pipefail

PY="${PY:-/home/zeus/miniconda3/envs/cloudspace/bin/python3}"
cd "$(dirname "$0")/../.."
mkdir -p runs/logs

# τ=0.10 的 lr 沿用 E29 定出的 0.03；τ=0.28 的由 R1 定出，預設值僅為佔位。
LR_010="${LR_010:-0.03}"
LR_028="${LR_028:-0.03}"

BASE="--sites P --ranks 16 --size 512 --steps 150 --k_inv 10 --n_edit 10 \
  --limit 2 --guidance_scale 7.5 --beta_linf 0 --margin 1.0 --alpha_lpips 0 \
  --stop_on_plateau"

{
  echo "=== E31 R2 主網格開始 $(date -Is) ==="
  echo "LR_010=$LR_010  LR_028=$LR_028"
  "$PY" -c "import torch; print('torch', torch.__version__, 'cuda', torch.cuda.is_available())"

  for TAU in 0.10 0.28; do
    if [ "$TAU" = "0.10" ]; then LR="$LR_010"; else LR="$LR_028"; fi
    for S in 0.5 0.3; do
      "$PY" scripts/run_defense.py $BASE --tau_lpips "$TAU" --lr "$LR" \
        --strength "$S" --defense_mode untargeted \
        --out "runs/e31_untargeted_tau${TAU}_s${S}"

      "$PY" scripts/run_defense.py $BASE --tau_lpips "$TAU" --lr "$LR" \
        --strength "$S" --defense_mode targeted \
        --target_image data/targets/gray.png \
        --out "runs/e31_targeted_tau${TAU}_s${S}"

      "$PY" scripts/run_defense.py $BASE --tau_lpips "$TAU" --lr "$LR" \
        --strength "$S" --defense_mode crossattn --attn_mode suppress \
        --attn_timesteps 4 --out "runs/e31_attn_tau${TAU}_s${S}"
    done
  done

  echo "=== 綁定者判定 ==="
  "$PY" scripts/e27_binding_check.py runs/e31_*_tau*_s*
  echo "=== E31 R2 主網格結束 $(date -Is) ==="
} 2>&1 | tee runs/logs/e31_grid.log
```

```python
"""E31 主網格的彙整與 ISR 判定。

判定邏輯不在此處重寫，直接呼叫 p12 的 `judge_cell`：同一個判準在兩個地方
各寫一次，遲早會分歧，而本專案的結論全部建立在判準上。

執行：python scripts/e31_report.py --degrade_tau <p11 定出的值>
成本：只讀 CSV，秒級。
"""

import argparse
import csv
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parent.parent
sys.path.insert(0, str(ROOT))

from scripts.p12_isr_rejudge import collect, judge_cell


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--prefix", default="e31_")
    ap.add_argument("--degrade_metric", default="niqe")
    ap.add_argument("--degrade_tau", type=float, required=True,
                    help="p11 的人眼階梯定出的劣化門檻。必填——用預設值跑出來"
                         "的判定沒有定錨，不可寫進報告")
    ap.add_argument("--out", default="runs/e31_report")
    args = ap.parse_args()

    runs = sorted(p for p in (ROOT / "runs").iterdir()
                  if p.is_dir() and p.name.startswith(args.prefix)
                  and (p / "results.csv").exists())
    if not runs:
        raise FileNotFoundError(
            f"找不到任何 {args.prefix}* 且含 results.csv 的 run。"
            "主網格尚未跑完，或跑的是 --no_eval"
        )
    out = ROOT / args.out
    out.mkdir(parents=True, exist_ok=True)

    rows = []
    for run in runs:
        cells = collect(run, args.degrade_metric)
        # 正式判準在乾淨攻擊那一格：無淨化、未見過的噪聲種子
        key = ("identity", "0.0", "heldout")
        if key not in cells:
            raise KeyError(
                f"{run.name} 沒有 (identity, 0.0, heldout) 這一格。"
                f"正式判準就在該格，缺席時不可用其他格代替"
            )
        sem, deg = cells[key]
        r = judge_cell(sem, deg, args.degrade_tau)
        # 收斂與否：stop_reason 為空代表用盡上限，該格不可用於比較
        with open(run / "summary.csv", encoding="utf-8") as f:
            srows = list(csv.DictReader(f))
        converged = all(s.get("stop_reason") for s in srows)
        rows.append({
            "run": run.name,
            "converged": converged,
            "steps_done": ",".join(s["steps_done"] for s in srows),
            "final_lpips": ",".join(s["final_lpips"] for s in srows),
            **r,
        })

    with open(out / "e31_summary.csv", "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=list(rows[0].keys()))
        w.writeheader()
        w.writerows(rows)

    print(f"{'run':<34} {'收斂':<5} {'ΔSigLIP':>9} {'Δ劣化':>8} "
          f"{'語意':<5} {'劣化':<5} ISR")
    for r in rows:
        print(f"{r['run']:<34} {'是' if r['converged'] else '否':<5} "
              f"{r['mean_dsiglip']:>+9.4f} {r['mean_ddegrade']:>+8.3f} "
              f"{'是' if r['semantic_fail'] else '否':<5} "
              f"{'是' if r['degrade_fail'] else '否':<5} "
              f"{'**是**' if r['isr'] else '否'}")
    n = sum(r["isr"] for r in rows)
    print(f"\n[e31] {len(rows)} 格中 ISR 成立 {n} 格")
    if n == 0:
        print("[e31] 正對照未建立。規格 §9 的第 1 種否定結果：預算與 strength "
              "兩個替代解釋都已排除")


if __name__ == "__main__":
    main()
```
